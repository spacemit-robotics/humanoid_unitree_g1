cmake_minimum_required(VERSION 3.16)
project(g1_policy_tests LANGUAGES CXX)
get_filename_component(SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(STAGING "${SDK_ROOT}/output/staging")
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
foreach(LIB rl behavior_manager robot_base operator_client)
    find_library(${LIB}_LIBRARY ${LIB} HINTS "${STAGING}/lib" NO_DEFAULT_PATH REQUIRED)
endforeach()
add_executable(test_policy_catalog test_policy_catalog.cpp)
target_compile_features(test_policy_catalog PRIVATE cxx_std_17)
target_include_directories(test_policy_catalog PRIVATE "${STAGING}/include"
    "${SDK_ROOT}/application/native/humanoid_common/src/behavior_manager")
target_link_libraries(test_policy_catalog PRIVATE Eigen3::Eigen yaml-cpp
    ${rl_LIBRARY} ${behavior_manager_LIBRARY} ${robot_base_LIBRARY})
add_executable(test_switch_client test_switch_client.cpp)
target_compile_features(test_switch_client PRIVATE cxx_std_17)
target_include_directories(test_switch_client PRIVATE "${STAGING}/include"
    "${SDK_ROOT}/application/native/humanoid_common/src/transport/transport_executor")
target_link_libraries(test_switch_client PRIVATE Eigen3::Eigen)
target_link_libraries(test_switch_client PRIVATE yaml-cpp ${operator_client_LIBRARY} Threads::Threads)
foreach(TEST test_policy_catalog test_switch_client)
    target_compile_options(${TEST} PRIVATE -Wall -Wextra -Werror)
    set_target_properties(${TEST} PROPERTIES BUILD_RPATH "${STAGING}/lib")
endforeach()
